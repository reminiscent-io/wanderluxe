8) /src/components/trip/travelers/schemas.ts (already added) — ensure this
import { z } from "zod";

export const travelerSchema = z.object({
  first_name: z.string().min(1, "First name is required"),
  last_name: z.string().optional(),
  shared_with_email: z
    .string()
    .email("Invalid email")
    .optional()
    .or(z.literal("").transform(() => undefined)),
  permission_level: z.enum(["edit", "read"]).default("read"),
});

export type TravelerForm = z.infer<typeof travelerSchema>;

with bad as (
  select 'accommodation' as kind, at.trip_id, at.stay_id as item_id, at.traveler_id
    from accommodation_travelers at
    left join trip_shares ts on ts.id = at.traveler_id and ts.trip_id = at.trip_id
   where ts.id is null
  union all
  select 'transportation', tt.trip_id, tt.transportation_id, tt.traveler_id
    from transportation_travelers tt
    left join trip_shares ts on ts.id = tt.traveler_id and ts.trip_id = tt.trip_id
   where ts.id is null
  union all
  select 'activity', dat.trip_id, dat.activity_id, dat.traveler_id
    from day_activity_travelers dat
    left join trip_shares ts on ts.id = dat.traveler_id and ts.trip_id = dat.trip_id
   where ts.id is null
  union all
  select 'reservation', rt.trip_id, rt.reservation_id, rt.traveler_id
    from reservation_travelers rt
    left join trip_shares ts on ts.id = rt.traveler_id and ts.trip_id = rt.trip_id
   where ts.id is null
)
